/* Tunggu maks 20 ms sampai frame keluar dari mailbox.
 * Return 1 = terkirim (ada yang ACK), 0 = tidak ada ACK (frame dibatalkan). */
static uint8_t CAN_WaitTxDone(uint32_t mailbox)
{
  uint32_t t0 = HAL_GetTick();

  while (HAL_CAN_IsTxMessagePending(&hcan, mailbox))
  {
    if ((HAL_GetTick() - t0) >= 20U)
    {
      HAL_CAN_AbortTxRequest(&hcan, mailbox);
      return 0;
    }
  }
  return 1;
}

/* Baca error counter dan pulihkan bus-off (AutoBusOff = DISABLE) */
static void CAN_ReadStatus(void)
{
  uint32_t esr = hcan.Instance->ESR;

  can_tec    = (uint8_t)((esr >> 16) & 0xFF);
  can_rec    = (uint8_t)((esr >> 24) & 0xFF);
  can_lec    = (uint8_t)((esr >> 4) & 0x07);
  can_busoff = (uint8_t)((esr >> 2) & 0x01);

  if (can_busoff)
  {
    HAL_CAN_Stop(&hcan);
    HAL_CAN_Start(&hcan);
    can_busoff_count++;
  }
}

/* Baca 1 channel ADC (polling) */
static uint32_t ADC_ReadChannel(uint32_t channel)
{
  uint32_t val;

  adcCfg.Channel      = channel;
  adcCfg.Rank         = ADC_REGULAR_RANK_1;
  adcCfg.SamplingTime = ADC_SAMPLETIME_55CYCLES_5;

  HAL_ADC_ConfigChannel(&hadc1, &adcCfg);
  HAL_ADC_Start(&hadc1);
  HAL_ADC_PollForConversion(&hadc1, 100);
  val = HAL_ADC_GetValue(&hadc1);
  HAL_ADC_Stop(&hadc1);

  return val;
}

/* Baca 1 NTC 10k (B=3950) dan hitung suhu.
 * Kalau ADC mendekati 0 atau 4095 (sensor putus/korslet) suhu diset NTC_INVALID_C
 * supaya tidak jadi NaN/inf. */
static void ReadNTC(uint32_t channel,
                    volatile uint32_t *adc, volatile float *v,
                    volatile float *r, volatile float *k, volatile float *c)
{
  *adc = ADC_ReadChannel(channel);
  *v   = ((float)(*adc) * 3.3f) / 4095.0f;

  if (*adc < 5U || *adc > 4090U)
  {
    *r = 0.0f;
    *k = NTC_INVALID_C + 273.15f;
    *c = NTC_INVALID_C;
    return;
  }

  *r = (10000.0f * (*v)) / (3.3f - (*v));
  *k = 1.0f / ((1.0f / 298.15f) + ((1.0f / 3950.0f) * logf((*r) / 10000.0f)));
  *c = (*k) - 273.15f;
}

  * @brief  The application entry point.
  * @retval int
  */

{

  /* USER CODE END 1 */

  /* MCU Configuration--------------------------------------------------------*/

  /* Reset of all peripherals, Initializes the Flash interface and the Systick. */

  /* USER CODE BEGIN Init */

  /* Configure the system clock */

  /* USER CODE BEGIN SysInit */

  /* USER CODE END SysInit */

  /* Initialize all configured peripherals */
  MX_GPIO_Init();
  MX_ADC1_Init();
  MX_CAN_Init();
  /* USER CODE BEGIN 2 */

  canFilter.FilterBank = 0;
  canFilter.FilterMode = CAN_FILTERMODE_IDMASK;
  canFilter.FilterScale = CAN_FILTERSCALE_32BIT;

  canFilter.FilterIdHigh = 0x0000;
  canFilter.FilterIdLow = 0x0000;

  canFilter.FilterMaskIdHigh = 0x0000;
  canFilter.FilterMaskIdLow = 0x0000;

  canFilter.FilterFIFOAssignment = CAN_FILTER_FIFO0;
  canFilter.FilterActivation = ENABLE;

  if (HAL_CAN_ConfigFilter(&hcan, &canFilter) != HAL_OK)
  {
    Error_Handler();
  }

  txHeader.IDE = CAN_ID_STD;
  txHeader.RTR = CAN_RTR_DATA;
  txHeader.TransmitGlobalTime = DISABLE;

  if (HAL_CAN_Start(&hcan) != HAL_OK)
  {
    Error_Handler();
  }

  /* Baud = PCLK1 / Prescaler / (1 + BS1 + BS2) = PCLK1 / Prescaler / (1 + 5 + 2) */
  can_pclk1     = HAL_RCC_GetPCLK1Freq();
  can_baud_calc = can_pclk1 / (hcan.Init.Prescaler * 8U);

  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */

  while (1)
  {
    /* ---- Baca semua sensor ---- */
    adc_value_voltage = ADC_ReadChannel(ADC_CHANNEL_6);
    voltage = ((float)adc_value_voltage * 3.3f) / 4095.0f;
    HAL_Delay(100);

    ReadNTC(ADC_CHANNEL_0, &adc_value_ntc1, &voltage_ntc1, &resistance_ntc1, &temperatureK_ntc1, &temperatureC_ntc1);
    HAL_Delay(100);

    ReadNTC(ADC_CHANNEL_1, &adc_value_ntc2, &voltage_ntc2, &resistance_ntc2, &temperatureK_ntc2, &temperatureC_ntc2);
    HAL_Delay(100);

    ReadNTC(ADC_CHANNEL_2, &adc_value_ntc3, &voltage_ntc3, &resistance_ntc3, &temperatureK_ntc3, &temperatureC_ntc3);
    HAL_Delay(100);

    ReadNTC(ADC_CHANNEL_3, &adc_value_ntc4, &voltage_ntc4, &resistance_ntc4, &temperatureK_ntc4, &temperatureC_ntc4);
    HAL_Delay(100);

    ReadNTC(ADC_CHANNEL_4, &adc_value_ntc5, &voltage_ntc5, &resistance_ntc5, &temperatureK_ntc5, &temperatureC_ntc5);
    HAL_Delay(100);

    /* ---- Susun data CAN ---- */
    voltage_mV = (uint16_t)(voltage * 1000.0f);
    temp1 = (int16_t)(temperatureC_ntc1 * 100.0f);
    temp2 = (int16_t)(temperatureC_ntc2 * 100.0f);
    temp3 = (int16_t)(temperatureC_ntc3 * 100.0f);
    temp4 = (int16_t)(temperatureC_ntc4 * 100.0f);
    temp5 = (int16_t)(temperatureC_ntc5 * 100.0f);
    ntc1_data = (uint16_t)temp1;
    ntc2_data = (uint16_t)temp2;
    ntc3_data = (uint16_t)temp3;
    ntc4_data = (uint16_t)temp4;
    ntc5_data = (uint16_t)temp5;

    canData1[0] = voltage_mV >> 8;
    canData1[1] = voltage_mV & 0xFF;
    canData1[2] = ntc1_data >> 8;
    canData1[3] = ntc1_data & 0xFF;
    canData1[4] = ntc2_data >> 8;
    canData1[5] = ntc2_data & 0xFF;
    canData1[6] = ntc3_data >> 8;
    canData1[7] = ntc3_data & 0xFF;

    canData2[0] = ntc4_data >> 8;
    canData2[1] = ntc4_data & 0xFF;
    canData2[2] = ntc5_data >> 8;
    canData2[3] = ntc5_data & 0xFF;

    /* ---- Kirim frame ---- */
    uint8_t tx_fail = 0;
    can_frame111_sent = 0;
    can_frame112_sent = 0;

    txHeader.StdId = 0x111;
    txHeader.DLC = 8;

    if (HAL_CAN_AddTxMessage(&hcan, &txHeader, canData1, &txMailbox) != HAL_OK)
    {
      tx_fail = 1;
      can_hal_error = HAL_CAN_GetError(&hcan);
      HAL_CAN_AbortTxRequest(&hcan, CAN_TX_MAILBOX0 | CAN_TX_MAILBOX1 | CAN_TX_MAILBOX2);
    }
    else
    {
      can_frame111_sent = CAN_WaitTxDone(txMailbox);
      if (can_frame111_sent) can_cnt_111++;
    }

    txHeader.StdId = 0x112;
    txHeader.DLC = 4;

    if (HAL_CAN_AddTxMessage(&hcan, &txHeader, canData2, &txMailbox) != HAL_OK)
    {
      tx_fail = 1;
      can_hal_error = HAL_CAN_GetError(&hcan);
      HAL_CAN_AbortTxRequest(&hcan, CAN_TX_MAILBOX0 | CAN_TX_MAILBOX1 | CAN_TX_MAILBOX2);
    }
    else
    {
      can_frame112_sent = CAN_WaitTxDone(txMailbox);
      if (can_frame112_sent) can_cnt_112++;
    }

    /* ---- Status akhir siklus ---- */
    if (tx_fail)
    {
      can_status = 3;
      can_cnt_fail++;
    }
    else if (can_frame111_sent && can_frame112_sent)
    {
      can_status = 1;
      can_cnt_sent_ok++;
    }
    else
    {
      can_status = 2;
      can_cnt_noack++;
    }

    CAN_ReadStatus();
    can_loop_count++;
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */

  }

  /* USER CODE END 3 */
}
